class Solution {
    public int minSteps(int n) {
        final int FACTOR = findFactor(n);
        
        if(n == 1) {
            return 0;
        }
        else if(FACTOR == n) {
            return n;
        }

        return minSteps(n / FACTOR) + FACTOR;
    }

    private int findFactor(int n) {
        if(n <= 1) {
            return n;
        }

        for(int i = 2; i <= Math.sqrt(n); ++i) {
            if(n % i == 0) {
                return i;
            }
        }
 
        return n;
    }
}
